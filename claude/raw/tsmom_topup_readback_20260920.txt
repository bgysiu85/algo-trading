import databento as db, pandas as pd, os, glob, collections, warnings
warnings.filterwarnings("ignore")
D=os.path.expanduser("~/mnt/GLBX.MDP3")
CYCLE={"GC":[2,4,6,8,10,12],"SI":[3,5,7,9,12],"HG":[3,5,7,9,12],"ZN":[3,6,9,12],"TN":[3,6,9,12]}
def maturities(root):
    m={}
    for f in sorted(glob.glob(f"{D}/definition/{root}/*.dbn.zst")):
        d=db.DBNStore.from_file(f).to_df()
        d=d[d.instrument_class=="F"]
        for iid,y,mo,sym,exp in zip(d.instrument_id,d.maturity_year,d.maturity_month,d.raw_symbol,d.expiration):
            m[int(iid)]=(int(y),int(mo),sym,exp)
    return m
def bars(root):
    fs=[f"{D}/ohlcv-1d/{root}.dbn.zst"]+([f"{D}/ohlcv-1d/{root}.c2-c4.dbn.zst"] if os.path.exists(f"{D}/ohlcv-1d/{root}.c2-c4.dbn.zst") else [])
    return pd.concat([db.DBNStore.from_file(f).to_df() for f in fs])
out=[]
for root,cyc in CYCLE.items():
    mat=maturities(root); b=bars(root)
    b["date"]=b.index.tz_convert(None).normalize() if b.index.tz is not None else b.index.normalize()
    b["mat"]=b.instrument_id.map(lambda i: mat.get(int(i),(None,None,None,None))[:2])
    unk=b.mat.map(lambda x:x[0] is None).mean()
    sess=sorted(b.date.unique())
    # first session of each month
    first={}
    for s in sess: first.setdefault((s.year,s.month),s)
    idx={s:i for i,s in enumerate(sess)}
    def held(t):
        y,m=t.year,t.month
        for k in range(0,30):
            yy,mm=y+(m-1+k)//12,(m-1+k)%12+1
            if mm not in cyc: continue
            fs=first.get((yy,mm))
            if fs is None: return (yy,mm)
            if idx[t] < idx[fs]-5: return (yy,mm)
        return None
    by=collections.defaultdict(set)
    vol={}
    for d,mt,sym,v in zip(b.date,b.mat,b.symbol,b.volume):
        by[d].add(mt); vol[(d,mt)]=v
    miss=[];vols=[]
    for s in sess:
        h=held(s)
        if h in by[s]: vols.append(vol[(s,h)])
        else: miss.append(s)
    out.append((root,len(sess),len(miss),f"{100*len(miss)/len(sess):.1f}%",int(pd.Series(vols).median()) if vols else 0,f"{unk:.1%}",str(sess[0].date()),[str(x.date()) for x in miss[:6]]))
print(pd.DataFrame(out,columns=["root","sessions","held_no_bar","pct","median_vol_held","unmapped_iid","first","first_misses"]).to_string(index=False))
